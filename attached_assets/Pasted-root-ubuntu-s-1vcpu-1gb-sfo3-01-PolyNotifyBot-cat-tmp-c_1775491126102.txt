from dotenv import load_dotenv
load_dotenv("/root/PolyNotifyBot/.env")

from py_clob_client.client import ClobClient
from py_clob_client.clob_types import ApiCreds, BalanceAllowanceParams, AssetType

private_key    = os.environ.get("POLY_PRIVATE_KEY", "")
proxy_address  = os.environ.get("POLY_PROXY_ADDRESS", "")
api_key        = os.environ.get("POLY_API_KEY", "")
api_secret     = os.environ.get("POLY_API_SECRET", "")
api_passphrase = os.environ.get("POLY_API_PASSPHRASE", "")

print(f"POLY_API_KEY (first 12)   : {api_key[:12]}...")

print("\n=== DERIVE API KEY FROM PRIVATE KEY ===")
try:
    derived = c0.derive_api_key()
    print(f"Derived key: {derived}")
except Exception as e:
    print(f"Error: {e}")

print("\n=== BALANCE WITH sig_type=0 (EOA) ===")
try:
    creds = ApiCreds(api_key=api_key, api_secret=api_secret, api_passphrase=api_passphrase)

    r = c2.get_balance_allowance(BalanceAllowanceParams(asset_type=AssetType.COLLATERAL, signature_type=-1))
    print(f"Result: {r}")
except Exception as e:
    print(f"Error: {e}")
