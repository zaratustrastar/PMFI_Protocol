#!/usr/bin/env python3
"""
FIX: Kalshi event→market resolution picks wrong sub-market

ROOT CAUSE: _kalshi_label_match_score only checks subtitle/title/sub_title.
Kalshi's API puts candidate names in `yes_sub_title` (e.g. "Marco Rubio").
The code never checks this field, so label matching scores 0 for every
sub-market and falls through to the alphabetical fallback — picking the
wrong candidate (e.g. TCAR=Ted Cruz instead of MRUB=Marco Rubio).

FIX 1: Add yes_sub_title and no_sub_title to _kalshi_label_match_score.
FIX 2: Add price-based validation in resolve_market_ticker — after label
        matching picks a candidate, verify its yes_bid_dollars matches
        Oddpool's expected price. Reject if delta > 30¢ (same approach
        Polymarket uses in _resolve_poly_tokens).

Apply:
  cd ~/PolyNotifyBot
  python3 fix_kalshi_resolution.py
  git diff
  git add -A && git commit -m "fix: kalshi resolution uses yes_sub_title + price validation"
  git push && sudo systemctl restart psniper-bot
"""

import sys, os

KALSHI_PATH = os.path.expanduser("~/PolyNotifyBot/bot/arb_monitor/adapters/kalshi.py")

def patch_file(filepath, replacements):
    with open(filepath, 'r') as f:
        content = f.read()
    for old, new in replacements:
        if old not in content:
            print(f"  ⚠️  Pattern not found in {filepath}:")
            print(f"      {repr(old[:120])}...")
            sys.exit(1)
        count = content.count(old)
        if count > 1:
            print(f"  ⚠️  Pattern appears {count} times (expected 1):")
            print(f"      {repr(old[:120])}...")
            sys.exit(1)
        content = content.replace(old, new)
        print(f"  ✅ Patched: {old[:70].strip()}...")
    with open(filepath, 'w') as f:
        f.write(content)
    print(f"  📝 Saved {filepath}")


fixes = []

# ═══════════════════════════════════════════════════════════════════════════════

fixes.append((

fixes.append((
