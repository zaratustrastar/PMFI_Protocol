#!/usr/bin/env bash
set -e

# Put config in user's home directory
INNGEST_CONFIG="$HOME/.config/inngest/inngest.yaml"

if [[ ! -f "${INNGEST_CONFIG}" ]]; then
  mkdir -p "$(dirname "${INNGEST_CONFIG}")"

  if [[ -n "${DATABASE_URL}" ]]; then
    # Use Postgres if DATABASE_URL is set
    cat > "${INNGEST_CONFIG}" <<EOF2
postgres-uri: "${DATABASE_URL}"
EOF2
  else
    # Fall back to SQLite in a local folder
    SQLITE_DIR="/opt/polymarket-bot/.local/share/inngest"
    mkdir -p "${SQLITE_DIR}"

    cat > "${INNGEST_CONFIG}" <<EOF2
sqlite-dir: "${SQLITE_DIR}"
EOF2
  fi
fi

# Run Inngest via npx so the local dependency is found
exec npx inngest-cli dev \
  -u http://localhost:5000/api/inngest \
  --host 127.0.0.1 \
  --port 3000 \
  --config "${INNGEST_CONFIG}"
