import requests, os
from dotenv import load_dotenv
from eth_hash.auto import keccak
load_dotenv()

rpc   = os.getenv('BASE_RPC_URL', 'https://mainnet.base.org')
vault = '0x1182054b82f96c110698eAe6Af73FdA013599F9d'
tx    = '0xb9d6e19512b27cef6abbd50ef98417cc115399fd6cb50976954b1eee9eb3e5c2'

def rpc_call(method, params):
    r = requests.post(rpc, json={'jsonrpc':'2.0','id':1,'method':method,'params':params}, timeout=10)
    return r.json()['result']

def view(sig):
    sel = keccak(sig.encode())[:4].hex()
    return rpc_call('eth_call', [{'to': vault, 'data': '0x'+sel}, 'latest'])

receipt = rpc_call('eth_getTransactionReceipt', [tx])
status  = int(receipt['status'], 16) if receipt else -1
print(f'setArbServicerWallet tx: {\"SUCCESS\" if status==1 else \"REVERTED\" if status==0 else \"PENDING\"}')

# Correct getter name is arbServicerWallet(), not servicerWallet()
sw   = '0x' + view('arbServicerWallet()')[-40:]
idle = int(view('idleBalance()'), 16) / 1e6
ltt  = int(view('lastTendTimestamp()'), 16)

import time
print(f'arbServicerWallet: {sw}')
print(f'idleBalance:       {idle:.6f} USDC')
print(f'lastTendTimestamp: {ltt}  (tend cooldown 60s)')
